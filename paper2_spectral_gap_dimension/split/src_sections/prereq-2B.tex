\subsection{Background imported from the core paper}\label{sec:prereq}

We use the following facts about the $e$-prototype
$Q_{j+1}=(4j+2)Q_j+Q_{j-1}$ proved in the companion core paper;
proofs are not repeated here.

\begin{theorem}[Gauge \label{thm:gauge}]
For $k\ge1$ let $h_k(1/Q_n)=(2k+1)^{-n}$.
Then $\{\Lam=k/4\}$ has infinite but $\sigma$-finite
$\mathcal{H}^{h_k}$-measure on $\R/\Z$, while
$\{\Lam<k/4\}$ is $\mathcal{H}^{h_k}$-null.
In the family $g_\beta(1/Q_n)=\beta^{-n}$ the level is null for
$\beta>2k+1$ and non-$\sigma$-finite for $1<\beta<2k+1$.
\end{theorem}

\begin{corollary}[Exact gauge \label{cor:exact-gauge}]
The gauge $h_k$ is exact: smaller gauges give measure zero even on
$\{\Lam\le k/4\}$, larger gauges destroy $\sigma$-finiteness on
$\{\Lam=k/4\}$, and the $\mathcal{H}^{h_k}$-measure of
$\{\Lam=k/4\}$ is infinite in every nonempty open arc.
\end{corollary}

\begin{corollary}[Infinite level \label{cor:infinite-level}]
$\{\Lam<\infty\}$ is Lebesgue-null, $\Lam=\infty$ almost everywhere on
every $[a,b]$, and $\dimH\{\Lam=\infty\}=1$.
\end{corollary}

\begin{theorem}[Subcritical dimensions \label{conj:weight}]
For $0\le\alpha<1$ the sublevel and exact-limit sets of
$\Lam_\alpha(x)=\limsup j^\alpha\|Q_jx\|$ have Hausdorff dimension
$1-\alpha$ (admissible targets $c\ge0$; $c\in[0,1/2]$ if $\alpha=0$).
\end{theorem}

\begin{corollary}[Gauge classification \label{cor:liminf-gauges}]
Let $\rho_b(g)=\liminf_n b^n g(1/Q_n)$ with $b=2k+1$ for
$S_k$ and $b=2$ for the unsigned exact limits.
The set is $\mathcal{H}^g$-null if $\rho_b(g)=0$;
$\sigma$-finite with infinite measure in every nonempty open set if
$0<\rho_b(g)<\infty$; non-$\sigma$-finite in every nonempty open set
if $\rho_b(g)=\infty$. For $b=2k+1$ nullity extends to
$\{\Lam\le k/4\}$.
\end{corollary}

\begin{theorem}[Trichotomy \label{thm:trichotomy}]
At each $k\ge1$ the limsup level (base $2k+1$), the unsigned exact limits
(universal binary base $2$), and the signed exact limits (each a single
countable $L$-coset) split three ways.
\end{theorem}
